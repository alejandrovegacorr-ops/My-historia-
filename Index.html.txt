    let selectedPerson = null;


    function showScreen(id) {

        document.querySelectorAll(".screen").forEach(screen => {
            screen.classList.remove("active");
        });

        document.getElementById(id).classList.add("active");

        window.scrollTo({
            top: 0,
            behavior: "smooth"
        });
    }


    function selectPerson(person) {

        selectedPerson = person;

        const data = messages[person];

        document.getElementById("questionTitle").textContent =
            person === "mama" ? "Para mamá" : "Para Yareli";

        document.getElementById("questionText").textContent =
            data.question;

        document.getElementById("answerInput").value = "";

        document.getElementById("errorText").textContent = "";

        document.getElementById("lockIcon").textContent = "🔒";

        showScreen("questionScreen");

        setTimeout(() => {
            document.getElementById("answerInput").focus();
        }, 500);
    }


    function normalize(text) {

        return text
            .toLowerCase()
            .normalize("NFD")
            .replace(/[\u0300-\u036f]/g, "")
            .replace(/[¿?¡!.,]/g, "")
            .trim();
    }


    function checkAnswer() {

        if (!selectedPerson) return;

        const input = normalize(
            document.getElementById("answerInput").value
        );

        const validAnswers =
            messages[selectedPerson].answers.map(normalize);

        if (validAnswers.includes(input)) {

            document.getElementById("errorText").textContent = "";

            document.getElementById("lockIcon").textContent = "🔓";

            showScreen("unlockScreen");

        } else {

            document.getElementById("errorText").textContent =
                "Esa no parece ser la respuesta... intenta recordar un poco más. ❤️";

            document.getElementById("answerInput").value = "";

            document.getElementById("answerInput").focus();
        }
    }


    function showLetter() {

        const data = messages[selectedPerson];

        document.getElementById("letterTitle").textContent =
            data.title;

        document.getElementById("letterBody").textContent =
            data.message;

        showScreen("letterScreen");
    }
